force_at(self: List<a>, n: Int) -> a

Returns head of a list or error

alist.force_at([1, 2, 3], 0) == 1
alist.force_at(["1"], 0) == "1"
alist.force_at([1, 2], 3) == error
alist.force_at([], 0) == error


force_head(self: List<a>) -> a

Returns head of a list or error

alist.force_head([1, 2, 3]) == 1
alist.force_head(["1"]) == "1"
alist.force_head([]) == error


force_tail(self: List<a>) -> List<a>

Returns tail of a list or empty list in case there is no tail.

alist.force_tail([1, 2, 3]) == [2, 3]
alist.force_tail(["1"] ) == []
alist.force_tail([]) == []


reduce_left(self: List<a>, with: fn(b, a) -> b, zero: b) -> b

Reduce a list from left to right.

Deprecated: please use list.reduce from stdlib.

alist.reduce_left([1, 2, 3], fn(n, total) { n + total }, 0) == 6
alist.reduce_left(["1", "2", "3"], fn(a, b) { builtin.append_string(a, b), "0" } ) == "0123"
alist.reduce_left([#[1], #[2], #[3]], fn(left, right) { bytearray.concat(left, right) }, #[9]) == #[9, 1, 2, 3]


reduce_right(self: List<a>, with: fn(b, a) -> b, zero: b) -> b

Reduce a list from right to left.

alist.reduce_right([1, 2, 3], fn(n, total) { n + total }, 0) == 6
alist.reduce_right(["1", "2", "3"], fn(a, b) { builtin.append_string(a, b) },  "0" ) == "0321"


first(self: List<a>) -> Option<a>

Returns first element from the list. This is just an alias to std lib’s head function.

non_empty(self: List<a>) -> Bool

Returns whether this list is NOT empty.

alist.non_empty([1, 2, 3]) == True


prepend(self: List<a>, item: a) -> List<a>

Prepends an item to the beginning of the list. This is just an alias to std lib’s push function Notice this operation is not cheap on functional lists.

alist.prepend([1, 2, 3], 0) == [0, 1, 2, 3]


append(self: List<a>, item: a) -> List<a>

Appends an element to the end of the list. This operation is very cheap on functional lists.

alist.append([1, 2, 3], 0) == [0, 1, 2, 3]


take_until(self: List<a>, predicate: fn(a) -> Bool) -> List<a>

Takes elements from the list until certain predicate is matched.

alist.take_until([1, 2, 3, 4, 5], fn(x) { x < 4 }) == [1, 2, 3]


reject(self: List<a>, predicate: fn(a) -> Bool) -> List<a>

Filters out elements matching a given predicate.

indexed_foldl(
  self: List<a>,
  zero: result,
  with: fn(Int, a, result) -> result,
) -> result

Like foldl, but also provides the position (0-based) of the elements when iterating.

let group = fn(i, x, xs) { [(i, x), ..xs] }
list.indexed_foldl(["a", "b", "c"], [], group) == [
  (2, "c"),
  (1, "b"),
  (0, "a")
]

let group = fn(i, xs, x) { [(i, x), ..xs] }
